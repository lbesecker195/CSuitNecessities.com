{{- /* Every in-content image becomes an affiliate hit target. Markdown syntax: ![alt](img.png) -> links to the page's `offer` product ![alt](img.png "Caption text") -> caption + links to the page's `offer` ![alt](img.png "https://amazon.com/... | Caption") -> explicit link + caption */ -}} {{- $src := .Destination -}} {{- $title := .Title -}} {{- $link := "" -}} {{- $caption := $title -}} {{- if strings.Contains $title "|" -}} {{- $parts := split $title "|" -}} {{- $link = trim (index $parts 0) " " -}} {{- $caption = trim (index $parts 1) " " -}} {{- else if or (strings.HasPrefix $title "http://") (strings.HasPrefix $title "https://") -}} {{- $link = trim $title " " -}}{{- $caption = "" -}} {{- end -}} {{- $offer := "" -}} {{- with .Page.Params.offer -}}{{- $offer = index hugo.Data.offers (string .) -}}{{- end -}} {{- /* No explicit link -> use the page's configured product. */ -}} {{- if and (not $link) $offer -}}{{- $link = partial "amzn.html" $offer -}}{{- end -}} {{- /* Explicit link that points at an offer domain -> tag it as that product. */ -}} {{- if $link -}} {{- $lu := urls.Parse $link -}} {{- if strings.Contains $lu.Hostname "amazon." -}} {{- if not (strings.Contains $link "tag=") -}} {{- $sep := cond (strings.Contains $link "?") "&" "?" -}} {{- $link = printf "%s%stag=%s" $link $sep (or site.Params.amazonTag "REPLACE-WITH-YOUR-AMAZON-TAG") -}} {{- end -}} {{- else -}} {{- range $k, $v := hugo.Data.offers -}}{{- if and $v.domain (in $lu.Hostname $v.domain) -}}{{- $offer = $v -}}{{- end -}}{{- end -}} {{- end -}} {{- end -}} {{- /* Resolve + resize the image. */ -}} {{- $img := .Page.Resources.GetMatch $src -}} {{- if not $img -}}{{- $img = resources.Get $src -}}{{- end -}} {{- $rendered := $src | relURL -}} {{- $w := 0 -}}{{- $h := 0 -}} {{- with $img -}} {{- $raster := in (slice "jpeg" "png" "webp" "tiff" "bmp") .MediaType.SubType -}} {{- $r := . -}} {{- if and $raster (gt .Width 1280) -}}{{- $r = .Resize "1280x webp q82" -}}{{- end -}} {{- $rendered = $r.RelPermalink -}} {{- if $raster -}}{{- $w = $r.Width -}}{{- $h = $r.Height -}}{{- end -}} {{- end -}} {{- $name := "" -}}{{- with $offer -}}{{- $name = .name -}}{{- end -}} {{- $overlay := or .Page.Params.imageOverlayText site.Params.imageOverlayText -}}
